cmake_minimum_required(VERSION 3.20)
project(tfire VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

  set(CMAKE_BUILD_TYPE Release)

option(TFIRE_SDL_STATIC "build SDL2 from source and link it statically" ON)

# ------------------------------------------------------------------ the platform id the updater asks for
if(TFIRE_PLATFORM STREQUAL "")
  string(TOLOWER "${CMAKE_SYSTEM_NAME}" _os)
  if(_os STREQUAL "darwin")
    set(_os "macos")
  endif()
  set(_arch "${CMAKE_SYSTEM_PROCESSOR}")
  if(CMAKE_OSX_ARCHITECTURES)
    set(_arch "${CMAKE_OSX_ARCHITECTURES}")
  endif()
  if(_arch MATCHES "^(x86_64|AMD64|amd64|x64)$")
    set(_arch "x64")
  elseif(_arch MATCHES "^(aarch64|arm64|ARM64)$")
    set(_arch "arm64")
  endif()
  set(TFIRE_PLATFORM "${_os}-${_arch}")
endif()
message(STATUS "tFire ${TFIRE_VERSION} for ${TFIRE_PLATFORM}")

# ------------------------------------------------------------------ files embedded as C arrays
function(tf_embed out_c symbol input)
  file(READ "${input}" _hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
  file(WRITE "${out_c}" "/* generated from ${input} */\nconst unsigned char ${symbol}[] = {\n${_bytes}0x00\n};\nconst unsigned long ${symbol}_len = sizeof(${symbol}) - 1;\n")
endfunction()
set(GEN "${CMAKE_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
tf_embed("${GEN}/prelude_js.c" tf_prelude "${CMAKE_SOURCE_DIR}/engine/prelude.js")
tf_embed("${GEN}/notices_html.c" tfire_notices "${CMAKE_SOURCE_DIR}/THIRD_PARTY_NOTICES.html")
tf_embed("${GEN}/cacert_pem.c" tf_cacert "${CMAKE_SOURCE_DIR}/third_party/cacert/cacert.pem")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  engine/prelude.js THIRD_PARTY_NOTICES.html third_party/cacert/cacert.pem)

# ------------------------------------------------------------------ the libraries tFire's engine is made of
set(NS ${CMAKE_SOURCE_DIR}/third_party/netsurf)
function(tf_netsurf name)
  file(GLOB_RECURSE _src "${NS}/${name}/src/*.c")
  list(FILTER _src EXCLUDE REGEX "css_property_parser_gen|/test/|autogenerated-element-type\\.c")
  add_library(${name} STATIC ${_src})
  target_compile_definitions(${name} PRIVATE WITHOUT_ICONV_FILTER NDEBUG)
  target_compile_options(${name} PRIVATE -w -std=gnu99)
  target_include_directories(${name} PUBLIC ${NS}/${name}/include PRIVATE ${NS}/${name}/src)
endfunction()
tf_netsurf(libparserutils)
tf_netsurf(libwapcaplet)
tf_netsurf(libhubbub)
tf_netsurf(libcss)
target_include_directories(libhubbub PRIVATE ${NS}/libparserutils/include)
target_include_directories(libcss PRIVATE ${NS}/libparserutils/include ${NS}/libwapcaplet/include)
target_include_directories(libparserutils PRIVATE ${NS}/libparserutils/include)
target_link_libraries(libhubbub PUBLIC libparserutils)
target_link_libraries(libcss PUBLIC libparserutils libwapcaplet)

  FetchContent_MakeAvailable(SDL2)
  set(SDL_TARGET SDL2-static)
